.equ PORTB, 0x6000
.equ PORTA, 0x6001
.equ DDRB,  0x6002
.equ DDRA,  0x6003

; cold boot 
.equ COLDBOOTADDR, 0x0200
.equ COLDBOOT1, 0x65
.equ RESETS, 0x201

.equ TEMP, 0x0210 ; 16 bytes of temporary 'registers'

; display
.equ RS, 0b00000010
.equ RW, 0b00000100 ; low on write
.equ E,  0b00001000


; notes
.equ SOUND_ENABLE, 0b00000001 

.org 0xFF00
; Notes in an octave, from C5 to C6

;.byte 0xE6, 0xE0, 0xDA, 0xD2, 0xC8, 0xC1, 0xB8, 0xAC, 0xA1, 0x97, 0x8A, 0x7A

; whole notes in an octave from C5 to C6 (including C6
;     C     D     E     F     G     A     B     C
.byte 0xE7, 0xDB, 0xC9, 0xC1, 0xAD, 0x97, 0x7B, 0x6D

.org 0x8000

init:
	; setup the VIA chip
	LDA #0xff
	STA DDRA
	STA DDRB
	
	; check for cold boot (0x0200 should have 0x65, 0x02)
	LDA COLDBOOTADDR
	CMP #COLDBOOT1
	BEQ warmboot

coldboot:
	; set memory so that future resets will be warmboots
	LDA #COLDBOOT1
	STA COLDBOOTADDR
	
	; set resets 
	LDA #0xff
	STA RESETS
	
	; prompt the LCD to know we need to be in 4-bit mode.
	; function set
	LDA #0b00100000 ; 4 bit mode, 2 lines of text, 8x5 chars
	STA PORTB
	
	JSR f_waitForLcdReady
	LDA #0b00101000 
	STA PORTB
	LDA #0b00100000 ; 4 bit mode, 2 lines of text, 8x5 chars
	STA PORTB
	
	; now we fall into warmboot.
warmboot:
	; increment reset counter
	INC RESETS
	
	
	; function set
	LDA #0b00101000 ; 4 bit mode, 2 lines of text, 8x5 chars
	JSR f_SendLcdInstruction
	
	; Display on/off Control
	LDA #0b00001110 ; Display on, cursor on, blink on
	JSR f_SendLcdInstruction
	
	; Entry mode set
	LDA #0b00000110 ; Increment for next char, do not scroll.
	JSR f_SendLcdInstruction
	
	; Clear Display
	LDA #0b00000001 ; 
	JSR f_SendLcdInstruction
	
	; Return home
	LDA #0b00000010 ; 
	JSR f_SendLcdInstruction
	
	; Start sending data

main:
	LDA #0x00
	
loop:
	PHA 
	JSR f_printByte
	LDA #' '
    JSR	f_SendLcdChar
	PLA
	PHA
	JSR f_playNote
	JSR f_delay
	PLA
	
	ADC #0x01
	CMP #0x08
	BCC loop

	
	JSR f_delay
	
	; Clear Display
	LDA #0b00000001 ; 
	JSR f_SendLcdInstruction
	
	
	; silence audio
	LDA #0x00
	STA PORTA

end:
	BRA end

f_delay:
	LDA #0x00
inner_loop:
	ADC #0x01
	BCC inner_loop
	CLC
	RTS


f_playNote:
	; A - note
	CMP #0x08
	BCC playNote_good
	LDA #0x00
playNote_good:
	TAX
	LDA 0xFF00, X
	ORA #SOUND_ENABLE
	PHA
	JSR f_printByte
	; return home
	LDA #0b00000010 ; 
	JSR f_SendLcdInstruction
	PLA
	STA PORTA
	
	RTS

.include lib/lib.txt

.org 0xfffc
	; reset vector
.byte 0x00, 0x80, 0x00, 0x00
