; 50ms intervals. (50ms - 800ms)


.org 0x8000

.include init.txt

main:
	; setup timer interrupt
	
	
	LDA #0b11000100 ; enable T1 interrupt and shift register interrupt.
	STA INT_ENABLE
	
	LDA #0b01011000 ; enable continuous interrupts on T1.
					; enable the shift register to shift out on phi2 (system clock)
	STA AUX_CONTROL
	
	LDA #<50000 ; set counter to 50k clock cycles, 50ms.
	STA T1LOW
	LDA #>50000
	STA T1HIGH
	
	LDA #<50000 ; set counter to 50k clock cycles, 50ms.
	STA T1CLOW
	LDA #>50000
	STA T1CHIGH
	

	
	LDA #0x00
	STA TIMER
	LDA #0xFF ; so that the first note is played
	
	; setup shift register on the via chip for audio playback.
	
	
	
loop:
	SEI
	
	PHA
	; print the note number
	JSR f_printByte
	LDA #' '
	JSR f_SendLcdChar
	
	; print the note being sent (probably, check the datasheet
	LDA SHIFTREG
	JSR f_printByte
	
	; return home
	LDA #0b00000010 ; 
	JSR f_SendLcdInstruction
	
	PLA
	CLI
	BRA loop
	
end:
	LDA #0x00
	STA SHIFTREG
	BRA end
	

	
irqb:
	; check whether this is a timer interrupt
	PHA
	LDA INT_FLAG
	BIT #0b00000100
	BNE irqb_shift
	
	; we assume this is a timer interrupt.
	
	; increment the timer. check if it's time to stop playing
	INC TIMER
	LDA TIMER
	CMP #8 ; the timer is set up for 1/20th of a second.
	BEQ irqb_silence
	BCC irqb_end
	
irqb_advance:
	; 1 second has passed, increment A.
	PLA
	CLC
	ADC #0x01
	
	; Check to see if the song should be over
	CMP #0x31
	BEQ end 
	
	PHA
	
	; get the next note to play, and play it.
	TAX
	LDA 0xFF00, X
	STA SHIFTREG
	
	; reset timer.
	LDA #0x00
	STA TIMER
	
	BRA irqb_end


irqb_silence:
	; this is the last tick of the note, stop playing.
	LDA #0x00
	STA SHIFTREG
    ; fall through

	
irqb_end:
	; Read T1C-L low or write T1L-H high to clear the interrupt.
	LDA #0xFF
	LDA T1CLOW
	PLA	
	RTI
	
	
irqb_shift:
	; we have successfully shifted a byte to the shift register.
	; we need to pulse the clock on the shift register to update the value, and
	; clear the interrupt flag before returning.
	
	
	; clear interrupt:
	LDA #0b00000100 ; writing this to the flag register should clear only the shift interrupt.
	STA INT_FLAG
	
	; pulse the clock.
	LDA PORTA
	EOR #0b10000000
	STA PORTA
	EOR #0b10000000
	STA PORTA
	
	; get outta here!
	PLA
	RTI




.include lib.txt

.org 0xfffc
	; reset vector
.byte 0x00, 0x80, <irqb, >irqb
