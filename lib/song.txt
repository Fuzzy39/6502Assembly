; note format: 2 bytes per note.
; duration is measued in 
; 50ms intervals. (50ms - 800ms)


.org 0x8000

.include init.txt

main:
	; setup timer interrupt
	
	
	LDA #0b11000000 ; enable T1 interrupt
	STA INT_ENABLE
	
	LDA #0b01000000 ; enable continuous interrupts on T1.
	STA AUX_CONTROL
	
	LDA #<50000 ; set counter to 50k clock cycles, 50ms.
	STA T1LOW
	LDA #>50000
	STA T1HIGH
	
	LDA #<50000 ; set counter to 50k clock cycles, 50ms.
	STA T1CLOW
	LDA #>50000
	STA T1CHIGH
	

	
	LDA #0x00
	STA TIMER
	LDA #0xFF ; so that the first note is played
loop:
	SEI
	
	PHA
	; print the note number
	JSR f_printByte
	LDA #' '
	JSR f_SendLcdChar
	
	; print the note being sent (probably, check the datasheet
	LDA PORTA
	JSR f_printByte
	
	; return home
	LDA #0b00000010 ; 
	JSR f_SendLcdInstruction
	
	PLA
	CLI
	BRA loop
	
end:
	LDA #0x00
	STA PORTA
	BRA end
	

	
irqb:
	; we assume this is a timer interrupt.
	PHA
	; increment the timer. check if it's time to stop playing
	INC TIMER
	LDA TIMER
	CMP #8 ; the timer is set up for 1/20th of a second.
	BEQ irqb_silence
	BCC irqb_end
	
irqb_advance:
	; 1 second has passed, increment A.
	PLA
	CLC
	ADC #0x01
	
	; Check to see if the song should be over
	CMP #0x31
	BEQ end 
	
	PHA
	
	; get the next note to play, and play it.
	TAX
	LDA 0xFF00, X
	STA PORTA
	
	; reset timer.
	LDA #0x00
	STA TIMER
	
	BRA irqb_end


irqb_silence:
	; this is the last tick of the note, stop playing.
	LDA #0x00
	STA PORTA
    ; fall through

	
irqb_end:
	; Read T1C-L low or write T1L-H high to clear the interrupt.
	LDA #0xFF
	LDA T1CLOW
	PLA	
	RTI

.include lib.txt

.org 0xfffc
	; reset vector
.byte 0x00, 0x80, <irqb, >irqb
